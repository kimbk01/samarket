(() => {
  const r2 = (n) => Math.round(n * 10) / 10;
  const vis = (el) => { const cs = getComputedStyle(el); if (cs.visibility === 'hidden' || cs.display === 'none' || +cs.opacity === 0) return false; const r = el.getBoundingClientRect(); return r.width > 0 && r.height > 0; };
  const safeTop = 62;
  // background at a point: walk ancestors for first non-transparent bg
  const bgAt = (x, y) => { let el = document.elementFromPoint(x, y); const chain=[]; while (el) { const b = getComputedStyle(el).backgroundColor; if (b && b !== 'rgba(0, 0, 0, 0)' && b !== 'transparent') return b + ' @' + (el.className?.toString?.().slice(0,40)||el.tagName); el = el.parentElement; } return 'none'; };
  const LIMIT = safeTop + 64;
  const out = { path: location.pathname + location.search, safeTop, bg_status: bgAt(220, 10), bg_header: bgAt(6, safeTop + 26), bg_header_mid: bgAt(220, safeTop + 4) };
  // header bottom: scan y downward at x=2 for border
  const hdr = document.elementFromPoint(220, safeTop + 26)?.closest('header,[data-app-sticky-header],.sector-header-shell,.chat-header,[data-stores-home-header]');
  if (hdr) { const r = hdr.getBoundingClientRect(); const cs = getComputedStyle(hdr); out.header = { tag: hdr.tagName, cls: hdr.className.toString().slice(0, 90), dom: hdr.closest('[data-dibay-domain]')?.getAttribute('data-dibay-domain') || null, top: r2(r.top), h: r2(r.height), bb: cs.borderBottomWidth + ' ' + cs.borderBottomColor }; }
  const rows = [...document.querySelectorAll('.sector-header-bar,.chat-header,[data-stores-home-address-row],.sam-tier1-header__row')].filter(vis).filter(e=>e.getBoundingClientRect().top<LIMIT).map(e=>{const r=e.getBoundingClientRect(); const cs=getComputedStyle(e); return {cls:e.className.toString().slice(0,60), top:r2(r.top), h:r2(r.height), x:r2(r.left), w:r2(r.width), pad:cs.paddingLeft+'/'+cs.paddingRight};});
  out.rows = rows;
  const items = [];
  const seen = new Set();
  for (const el of document.querySelectorAll('body *')) {
    if (!vis(el)) continue; const r = el.getBoundingClientRect();
    if (r.top < safeTop - 2 || r.bottom > LIMIT || r.top > LIMIT) continue;
    const tag = el.tagName.toLowerCase();
    if (tag === 'a' || tag === 'button') {
      const svg = el.querySelector('svg'); const g = el.querySelector('.app-back-glyph');
      const ir = (svg||g)?.getBoundingClientRect();
      items.push({ k: 'btn', label: (el.getAttribute('aria-label')||el.innerText||'').trim().slice(0,22), x: r2(r.left), y: r2(r.top), w: r2(r.width), h: r2(r.height), cx: r2(r.left + r.width/2), cy: r2(r.top + r.height/2), icon: ir ? `${g?'glyph':'svg'} ${r2(ir.width)}x${r2(ir.height)} @${r2(ir.left)},${r2(ir.top)} cx=${r2(ir.left+ir.width/2)} cy=${r2(ir.top+ir.height/2)}` : null, cls: el.className.toString().slice(0,50) });
    }
    const own = [...el.childNodes].some(n => n.nodeType === 3 && n.textContent.trim());
    if (own && !seen.has(el)) { seen.add(el); const cs = getComputedStyle(el); const rg = document.createRange(); rg.selectNodeContents(el); const tr = rg.getBoundingClientRect();
      items.push({ k: 'text', t: el.innerText.trim().slice(0,24), fs: cs.fontSize, fw: cs.fontWeight, lh: cs.lineHeight, color: cs.color, x: r2(tr.left), y: r2(tr.top), w: r2(tr.width), h: r2(tr.height), cy: r2(tr.top+tr.height/2), cls: el.className.toString().slice(0,40) }); }
  }
  out.items = items;
  return JSON.stringify(out);
})()
